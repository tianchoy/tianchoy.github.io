{{- end }} {{- if $counters }} {{- end }} {{- else -}} {{- warnf "params.analytics.goatcounter.site 还没填，页脚统计不会显示" -}} {{- end -}} {{- else if eq $p "umami" -}} {{- $u := $a.umami | default dict -}} {{- $id := trim (string ($u.websiteId | default "")) " " -}} {{- $src := trim (string ($u.script | default "https://cloud.umami.is/script.js")) " " -}} {{- if $id }} {{- else -}} {{- warnf "params.analytics.umami.websiteId 还没填" -}} {{- end -}} {{- else if eq $p "ga4" -}} {{- $g := $a.ga4 | default dict -}} {{- $id := trim (string ($g.measurementId | default "")) " " -}} {{- if $id }} {{- else -}} {{- warnf "params.analytics.ga4.measurementId 还没填" -}} {{- end -}} {{- else if eq $p "cloudflare" -}} {{- $c := $a.cloudflare | default dict -}} {{- $token := trim (string ($c.token | default "")) " " -}} {{- if $token }} {{- $beacon := printf "{\"token\":\"%s\"}" $token }} {{- else -}} {{- warnf "params.analytics.cloudflare.token 还没填" -}} {{- end -}} {{- else if eq $p "busuanzi" }}
